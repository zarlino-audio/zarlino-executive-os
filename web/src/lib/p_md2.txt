/* Lightweight per-fence highlighter. KW sets chosen by the fence language so
   a ```ts fence tokenises differently than ```json. Multi-line block comments
   and strings carry correctly across lines via the inBlock flag. */
const SCODE_KW: Record<string, string[]> = {
  ts:  ["const","let","var","function","return","if","else","for","while","do","new","class","extends","implements","interface","type","enum","public","private","protected","readonly","static","import","export","from","default","async","await","try","catch","finally","throw","typeof","instanceof","as","of","in","this","super","null","undefined","true","false"],
  tsx: ["const","let","var","function","return","if","else","for","while","do","new","class","extends","interface","type","enum","public","private","import","export","from","default","async","await","try","catch","throw","typeof","as","of","in","this","null","undefined","true","false"],
  js:  ["const","let","var","function","return","if","else","for","while","do","new","class","extends","import","export","from","default","async","await","try","catch","finally","throw","typeof","instanceof","of","in","this","null","undefined","true","false"],
  jsx: ["const","let","var","function","return","if","else","for","while","do","new","class","extends","import","export","from","default","async","await","try","catch","throw","typeof","of","in","this","null","undefined","true","false"],
  python: ["def","class","return","if","elif","else","for","while","in","not","and","or","import","from","as","try","except","finally","raise","with","lambda","yield","pass","break","continue","None","True","False"],
  bash: ["if","then","else","elif","fi","for","do","done","case","esac","function","local","return","exit","export","echo","cd","source","in"],
  sh:   ["if","then","else","elif","fi","for","do","done","case","esac","function","return","exit","export","echo","in"],
  json: [], css: [], html: [], text: [], md: [], yaml: [], toml: [],
};
const SCODE_DFLT = SCODE_KW.ts!;

function highlightCode(code: string, lang: string): ReactNode[] {
  const kws = new Set<string>(SCODE_KW[lang] ?? SCODE_DFLT);
  const lines = code.split("\n");
  const out: ReactNode[] = [];
  let inBlock = false;
  const isWord = (c: string) => /[A-Za-z0-9_$]/.test(c);
  lines.forEach((ln, li) => {
    const spans: { t: string; c?: string }[] = [];
    let i = 0;
    const n = ln.length;
    const push = (t: string, c?: string) => {
      if (!t) return;
      const last = spans[spans.length - 1];
      if (c && last && last.c === c) { last.t += t; return; }
      if (!c && last && !last.c) { last.t += t; return; }
      spans.push({ t, c });
    };
    while (i < n) {
      const ch = ln[i];
      const nx = ln[i + 1];
      if (inBlock) {
        const close = ln.indexOf("*/", i);
        if (close === -1) { push(ln.slice(i), "tok-c"); i = n; break; }
        push(ln.slice(i, close + 2), "tok-c");
        inBlock = false; i = close + 2; continue;
      }
      if (ch === "/" && nx === "/") { push(ln.slice(i), "tok-c"); i = n; break; }
      if (ch === "#" && !(i > 0 && /[\w]/.test(ln[i - 1]))) { push(ln.slice(i), "tok-c"); i = n; break; }
      if (ch === "/" && nx === "*") {
        const close = ln.indexOf("*/", i + 2);
        if (close === -1) { push(ln.slice(i), "tok-c"); inBlock = true; i = n; break; }
        push(ln.slice(i, close + 2), "tok-c"); i = close + 2; continue;
      }
      if (ch === '"' || ch === "'" || ch === "`") {
        const q = ch; let j = i + 1;
        while (j < n) {
          if (ln[j] === "\\") { j += 2; continue; }
          if (ln[j] === q) break;
          j++;
        }
        if (j < n) { push(ln.slice(i, j + 1), "tok-str"); i = j + 1; continue; }
        push(ln.slice(i), "tok-str"); i = n; break;
      }
      if (/[0-9]/.test(ch) || (ch === "." && nx && /[0-9]/.test(nx))) {
        let j = i; while (j < n && /[0-9a-fA-FxX_]/.test(ln[j])) j++;
        push(ln.slice(i, j), "tok-num"); i = j; continue;
      }
      if (isWord(ch)) {
        let j = i; while (j < n && isWord(ln[j])) j++;
        const w = ln.slice(i, j);
        if (kws.has(w)) push(w, w === "true" || w === "false" || w === "null" || w === "undefined" ? "tok-bool" : "tok-kw");
        else if (j < n && ln[j] === "(") push(w, "tok-fn");
        else push(w);
        i = j; continue;
      }
      if (/[{}()[\]<>=+\-*/%!&|^~?:;,]/.test(ch)) push(ch, "tok-op");
      else push(ch);
      i++;
    }
    out.push(
      <span key={li} className="md-code-line">
        {spans.map((sp, si) => (sp.c ? <span key={si} className={sp.c}>{sp.t}</span> : <span key={si}>{sp.t}</span>))}
        {li < lines.length - 1 ? "\n" : ""}
      </span>,
    );
  });
  return out;
}
