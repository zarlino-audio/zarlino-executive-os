// ---- Markdown (safe + lightweight, P0/P1/P2-ready) -------------------------

function decodeEntities(s: string): string {
  return s
    .replace(/&quot;/g, '"')
    .replace(/&#39;|&apos;/g, "'")
    .replace(/&amp;/g, "&")
    .replace(/&lt;/g, "<")
    .replace(/&gt;/g, ">")
    .replace(/&nbsp;/g, " ")
    .replace(/&mdash;/g, "—")
    .replace(/&#x([0-9a-fA-F]+);/g, (_, h) => String.fromCodePoint(parseInt(h, 16)))
    .replace(/&#(\d+);/g, (_, d) => String.fromCodePoint(parseInt(d, 10)));
}

/* A stateless code fence block. Renders a header with the fence language pill
   and a Copy button that copies ONLY this block's code. Language detection also
   drives the lightweight syntax highlighter below. */
function MdCodeBlock({ lang, code }: { lang: string; code: string }) {
  const [copied, setCopied] = useState(false);
  return (
    <div className="md-codeblock">
      <div className="md-codeblock-head">
        <span className="md-lang">{lang || "text"}</span>
        <button
          type="button"
          aria-label="Copy code block"
          onClick={() => {
            void copyText(code).then((ok) => {
              if (ok) {
                setCopied(true);
                setTimeout(() => setCopied(false), 1400);
              }
            });
          }}
        >
          {copied ? "✓ copied" : "copy"}
        </button>
      </div>
      <pre>
        <code>{highlightCode(code, lang)}</code>
      </pre>
    </div>
  );
}

/* Inline segment renderer — backticks/bold/italic/links wrapped as spans. */
const EMAIL_LINK = /(https?:\/\/[^\s)>\]]+)/g;
function inlineNodes(text: string, keyBase: number): ReactNode[] {
  const nodes: ReactNode[] = [];
  const parts = text.split(/(`[^`\n]+`|\*\*[^*\n]+\*\*|[*_][^*\n]+[*_])/g);
  parts.forEach((part, idx) => {
    if (!part) return;
    if (part.startsWith("`") && part.endsWith("`") && part.length > 2) {
      nodes.push(<code key={`${keyBase}-${idx}`}>{part.slice(1, -1)}</code>);
    } else if (part.startsWith("**") && part.endsWith("**") && part.length > 4) {
      nodes.push(<strong key={`${keyBase}-${idx}`}>{part.slice(2, -2)}</strong>);
    } else if (/^[*_]/.test(part) && /[*_]$/.test(part) && part.length > 2 && !part.includes("://")) {
      const inner = part.slice(1, -1);
      const push = inner.split((`[` + `${inner}`) as never); // split by '[' if linked
      nodes.push(<em key={`${keyBase}-${idx}`}>{inner.split("[")[0]}</em>);
      // catch any trailing [text](url) remainder inside emphas
      const rest = liftLinks(inner.split("[")[1] || "", `${keyBase}-${idx}-r`);
      if (rest) nodes.push(...rest);
      void push;
    } else {
      nodes.push(...liftLinks(part, `${keyBase}-${idx}`));
    }
  });
  return nodes;
}

/* Find [text](url) and bare urls; render <a> (target blank) for each. */
function liftLinks(part: string, keyBase: string): ReactNode[] {
  const out: ReactNode[] = [];
  const re = /!?\[([^\]]*)\]\(([^)\s]+)\)/;
  let rest = part;
  let ci = 0;
  for (;;) {
    const m = re.exec(rest);
    if (!m) break;
    if (m.index > 0) out.push(<span key={`${keyBase}-${ci}`}>{rest.slice(0, m.index)}</span>);
    const isImg = rest.startsWith("!", m.index - 1) === false && part[m.index] === "[" ? false : m[0].startsWith("![");
    void isImg;
    out.push(
      m[0].startsWith("![")
        ? <img key={`${keyBase}-${ci}i`} src={m[2]} alt={m[1]} loading="lazy" />
        : <a key={`${keyBase}-${ci}a`} href={m[2]} target="_blank" rel="noreferrer noopener">{m[1]}</a>,
    );
    rest = rest.slice(m.index + m[0].length);
    ci++;
  }
  if (rest) {
    const bits = rest.split(EMAIL_LINK);
    bits.forEach((b, i) => {
      if (!b) return;
      if (/^https?:\/\//.test(b))
        out.push(<a key={`${keyBase}-${ci}-${i}u`} href={b} target="_blank" rel="noreferrer noopener">{b}</a>);
      else out.push(<span key={`${keyBase}-${ci}-${i}`}>{b}</span>);
    });
  }
  return out;
}
