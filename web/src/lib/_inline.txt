/* Single-pass inline scanner: recognises (in priority order per char)
   ```code```/`code`, ![alt](url), [text](url), **bold**, *italic*/_italic_,
   and bare http(s) URLs; everything else is plain text. Emits React nodes.
   Safe because React escapes text and attribute values. */
function inlineNodes(text: string, keyBase: number): ReactNode[] {
  const nodes: ReactNode[] = [];
  let buf = "";
  const flush = (k: number) => {
    if (!buf) return;
    const bits = buf.split(/(https?:\/\/[^\s)]+)/g);
    bits.forEach((b, i) => {
      if (!b) return;
      if (/^https?:\/\//.test(b))
        nodes.push(<a key={`auto-${k}-${i}`} href={b} target="_blank" rel="noreferrer noopener">{b}</a>);
      else nodes.push(<span key={`${keyBase}-${k}-t${i}`}>{b}</span>);
    });
    buf = "";
  };
  let i = 0;
  let key = 0;
  const n = text.length;
  while (i < n) {
    const ch = text[i];
    // code span
    if (ch === "`") {
      const j = text.indexOf("`", i + 1);
      if (j > i) { flush(key); nodes.push(<code key={`${keyBase}-${key}-c`}>{text.slice(i + 1, j)}</code>); i = j + 1; key++; continue; }
    }
    // markdown link or image
    if ((ch === "!" && text[i + 1] === "[") || ch === "[") {
      const linkStart = ch === "[" ? i : i + 1;
      const toClose = text.indexOf("]", linkStart + 1);
      if (toClose > linkStart) {
        const openParen = text.indexOf("(", toClose + 1);
        const closeParen = text.indexOf(")", openParen + 1);
        if (openParen === toClose + 1 && closeParen > openParen) {
          const label = text.slice(linkStart + 1, toClose);
          const url = text.slice(openParen + 1, closeParen);
          flush(key);
          if (ch === "!") nodes.push(<img key={`${keyBase}-${key}-img`} src={url} alt={label} loading="lazy" />);
          else nodes.push(<a key={`${keyBase}-${key}-a`} href={url} target="_blank" rel="noreferrer noopener">{label}</a>);
          i = closeParen + 1; key++; continue;
        }
      }
    }
    // bold **x**
    if (ch === "*" && text[i + 1] === "*") {
      const j = text.indexOf("**", i + 2);
      if (j > i) { flush(key); nodes.push(<strong key={`${keyBase}-${key}-b`}>{text.slice(i + 2, j)}</strong>); i = j + 2; key++; continue; }
    }
    // emphasis *x* or _x_  (single char delimiter not adjacent to a letter both sides)
    if ((ch === "*" || ch === "_") && text[i + 1] && text[i + 1] !== ch) {
      const delim = ch;
      const end = text.indexOf(delim, i + 1);
      const isBold = text[i + 1] === delim; // handled above
      if (!isBold && end > i) {
        const before = i === 0 ? " " : text[i - 1];
        const after = end + 1 < n ? text[end + 1] : " ";
        const noWordBoundary = /[A-Za-z0-9]/.test(before) && /[A-Za-z0-9]/.test(after);
        if (!noWordBoundary) {
          flush(key);
          nodes.push(<em key={`${keyBase}-${key}-e`}>{text.slice(i + 1, end)}</em>);
          i = end + 1; key++; continue;
        }
      }
    }
    buf += ch;
    i++;
  }
  flush(key);
  return nodes;
}
